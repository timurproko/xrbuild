cmake_minimum_required(VERSION 3.21)
project(xrb_input_hand_tracking LANGUAGES CXX)
set(XRB_PLUGIN_SDK "$ENV{XRB_CORE}/native/plugin-sdk" CACHE PATH "XR Build plugin SDK")
set(XRB_CORE_THIRD_PARTY "$ENV{XRB_CORE}/third_party" CACHE PATH "Core header-only dependencies")
set(XRB_PLUGIN_BIN_DIR "" CACHE PATH "Explicit user-owned plugin installation folder")
if(NOT XRB_PLUGIN_BIN_DIR)
  message(FATAL_ERROR "Build through the panel or build.mjs with a selected writable workspace; product source is read-only")
endif()
foreach(required "${XRB_PLUGIN_SDK}/include/xrb_plugin.h" "${XRB_CORE_THIRD_PARTY}/nlohmann/nlohmann/json.hpp")
  if(NOT EXISTS "${required}")
    message(FATAL_ERROR "Missing ${required}; build via the panel or set XRB_CORE / XRB_PLUGIN_SDK / XRB_CORE_THIRD_PARTY")
  endif()
endforeach()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/samples/hand-tracking/fixtures/hand.json")
file(READ "samples/hand-tracking/fixtures/hand.json" HAND_FIXTURE_JSON)
configure_file(src/fixture.h.in "${CMAKE_CURRENT_BINARY_DIR}/fixture.h" @ONLY)
add_library(hand_tracking_core STATIC src/synthetic.cpp)
target_compile_features(hand_tracking_core PUBLIC cxx_std_17)
target_include_directories(hand_tracking_core PUBLIC src "${XRB_PLUGIN_SDK}/include" PRIVATE "${XRB_CORE_THIRD_PARTY}/nlohmann" "${CMAKE_CURRENT_BINARY_DIR}")
set_property(TARGET hand_tracking_core PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(xrb_input_hand_tracking SHARED src/plugin.cpp)
target_compile_definitions(xrb_input_hand_tracking PRIVATE XRB_PLUGIN_BUILD NOMINMAX)
target_link_libraries(xrb_input_hand_tracking PRIVATE hand_tracking_core)
set_property(TARGET xrb_input_hand_tracking PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_custom_command(TARGET xrb_input_hand_tracking POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E make_directory "${XRB_PLUGIN_BIN_DIR}"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:xrb_input_hand_tracking>" "${XRB_PLUGIN_BIN_DIR}/"
  VERBATIM)
include(CTest)
if(BUILD_TESTING)
  add_executable(hand-tracking-tests tests/plugin_tests.cpp)
  target_compile_definitions(hand-tracking-tests PRIVATE NOMINMAX)
  target_link_libraries(hand-tracking-tests PRIVATE hand_tracking_core)
  set_property(TARGET hand-tracking-tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_dependencies(hand-tracking-tests xrb_input_hand_tracking)
  add_test(NAME hand-tracking-tests COMMAND hand-tracking-tests "$<TARGET_FILE:xrb_input_hand_tracking>")
endif()
