cmake_minimum_required(VERSION 3.21)
project(hand_tracking_rgbd LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This bridge is currently qualified only on Windows x64")
endif()
if(NOT SENSOR_INCLUDE OR NOT SENSOR_LIBRARY OR NOT SENSOR_RUNTIME OR NOT SENSOR_CPP_HEADER OR NOT SENSOR_GEOMETRY_HEADER OR NOT XRB_CORE_THIRD_PARTY)
  message(FATAL_ERROR "Set the verified optional sensor dependency through bridge/build.mjs")
endif()
if(NOT MARKER_SOURCE)
  message(FATAL_ERROR "Prepare the pinned marker detector through bridge/build.mjs")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_PYTHON_WRAPPER OFF CACHE BOOL "" FORCE)
add_subdirectory("${MARKER_SOURCE}" "${CMAKE_BINARY_DIR}/marker-library" EXCLUDE_FROM_ALL)
add_executable(hand-tracking-rgbd capture.cpp)
target_link_libraries(hand-tracking-rgbd PRIVATE apriltag)
target_compile_features(hand-tracking-rgbd PRIVATE cxx_std_17)
target_compile_definitions(hand-tracking-rgbd PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(hand-tracking-rgbd PRIVATE /W4 /EHsc)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/sensor-headers.hpp" "#include \"${SENSOR_CPP_HEADER}\"\n#include \"${SENSOR_GEOMETRY_HEADER}\"\n")
target_include_directories(hand-tracking-rgbd PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${SENSOR_INCLUDE}" "${XRB_CORE_THIRD_PARTY}/nlohmann")
target_link_libraries(hand-tracking-rgbd PRIVATE "${SENSOR_LIBRARY}")
add_custom_command(TARGET hand-tracking-rgbd POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${SENSOR_RUNTIME}" "$<TARGET_FILE_DIR:hand-tracking-rgbd>/")
add_custom_command(TARGET hand-tracking-rgbd POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${MARKER_SOURCE}/LICENSE.md" "$<TARGET_FILE_DIR:hand-tracking-rgbd>/apriltag-LICENSE.md")
enable_testing()
add_test(NAME bridge-native-self-test COMMAND hand-tracking-rgbd --self-test)
add_test(NAME bridge-sdk-geometry COMMAND hand-tracking-rgbd --geometry-vectors)
add_test(NAME bridge-marker-self-test COMMAND hand-tracking-rgbd --marker-self-test)
