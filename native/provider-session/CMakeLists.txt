cmake_minimum_required(VERSION 3.21)
project(xrb_provider_session LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Native provider sessions currently support Windows only")
endif()
set(XRB_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." CACHE PATH "Core source root")
set(XRB_THIRD_PARTY "${XRB_ROOT}/third_party" CACHE PATH "Core native dependency root")
function(acquisition_target target source)
  add_executable(${target} ${source})
  target_compile_features(${target} PRIVATE cxx_std_20)
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0602)
  target_include_directories(${target} PRIVATE "${XRB_THIRD_PARTY}/nlohmann")
  target_link_libraries(${target} PRIVATE advapi32 bcrypt)
  set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
endfunction()
acquisition_target(xrb-acquisition acquisition_main.cpp)
include(CTest)
if(BUILD_TESTING)
  acquisition_target(acquisition-tests tests/acquisition_tests.cpp)
  acquisition_target(xrb-acquisition-test acquisition_main.cpp)
  target_compile_definitions(xrb-acquisition-test PRIVATE XRB_ACQUISITION_TEST=1)
  add_test(NAME acquisition-lease COMMAND acquisition-tests)
endif()
