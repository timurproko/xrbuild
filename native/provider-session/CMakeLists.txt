cmake_minimum_required(VERSION 3.21)
project(xrb_provider_session LANGUAGES C CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Native provider sessions currently support Windows only")
endif()
set(XRB_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." CACHE PATH "Core source root")
set(XRB_THIRD_PARTY "${XRB_ROOT}/third_party" CACHE PATH "Core native dependency root")
function(acquisition_target target source)
  add_executable(${target} ${source})
  target_compile_features(${target} PRIVATE cxx_std_20)
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0602)
  target_include_directories(${target} PRIVATE "${XRB_THIRD_PARTY}/nlohmann")
  target_link_libraries(${target} PRIVATE advapi32 bcrypt)
  set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
endfunction()
acquisition_target(xrb-acquisition acquisition_main.cpp)
set(HOST_COMMON "${CMAKE_CURRENT_SOURCE_DIR}/../host/src")
add_library(provider-common STATIC input_registry.cpp
  "${HOST_COMMON}/plugins.cpp" "${HOST_COMMON}/providers.cpp" "${HOST_COMMON}/builtin_inputs.cpp"
  "${HOST_COMMON}/host_channel.cpp" "${HOST_COMMON}/hand_recording.cpp" "${HOST_COMMON}/log.cpp")
target_compile_features(provider-common PUBLIC cxx_std_20)
target_compile_definitions(provider-common PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN JSON_DIAGNOSTICS=1)
target_include_directories(provider-common PUBLIC "${HOST_COMMON}" "${XRB_ROOT}/native/plugin-sdk/include" "${XRB_THIRD_PARTY}/nlohmann")
target_link_libraries(provider-common PUBLIC ws2_32 bcrypt shell32 ole32 advapi32)
acquisition_target(xrb-provider-session main.cpp)
target_link_libraries(xrb-provider-session PRIVATE provider-common)
include(CTest)
if(BUILD_TESTING)
  acquisition_target(acquisition-tests tests/acquisition_tests.cpp)
  acquisition_target(xrb-acquisition-test acquisition_main.cpp)
  target_compile_definitions(xrb-acquisition-test PRIVATE XRB_ACQUISITION_TEST=1)
  add_test(NAME acquisition-lease COMMAND acquisition-tests)
  acquisition_target(xrb-provider-session-test main.cpp)
  target_compile_definitions(xrb-provider-session-test PRIVATE XRB_PROVIDER_TEST=1)
  target_link_libraries(xrb-provider-session-test PRIVATE provider-common)
  acquisition_target(provider-frame-tests tests/frame_tests.cpp)
  target_link_libraries(provider-frame-tests PRIVATE provider-common)
  add_test(NAME provider-frames COMMAND provider-frame-tests)
  add_library(session-test-input SHARED "${XRB_ROOT}/native/plugin-sdk/tests/test_plugin.c")
  target_include_directories(session-test-input PRIVATE "${XRB_ROOT}/native/plugin-sdk/include")
  target_compile_definitions(session-test-input PRIVATE TP_ROLE=1 TP_NAME="session-test" TP_ABI_MAJOR=1 _CRT_SECURE_NO_WARNINGS)
  set_target_properties(session-test-input PROPERTIES PREFIX "" OUTPUT_NAME "xrb_session_test" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
endif()
