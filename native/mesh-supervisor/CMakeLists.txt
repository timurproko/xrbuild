cmake_minimum_required(VERSION 3.21)
project(xrb_mesh_supervisor LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Mesh process containment currently supports Windows only")
endif()
add_executable(xrb-mesh-supervisor main.cpp)
target_compile_features(xrb-mesh-supervisor PRIVATE cxx_std_20)
target_compile_definitions(xrb-mesh-supervisor PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
set_target_properties(xrb-mesh-supervisor PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}" MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
